CREATE DATABASE IF NOT EXISTS 4r6_hays 
  CHARACTER SET utf8mb4 
  COLLATE utf8mb4_general_ci;
USE 4r6_hays;


DROP TABLE IF EXISTS rfid_logs;
DROP TABLE IF EXISTS rfid_registered;


CREATE TABLE rfid_registered (
  id INT AUTO_INCREMENT PRIMARY KEY,
  rfid_tag VARCHAR(20) NOT NULL UNIQUE,
  status TINYINT(1) NOT NULL DEFAULT 0,
  last_update DATETIME DEFAULT CURRENT_TIMESTAMP 
               ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


CREATE TABLE rfid_logs (
  id INT AUTO_INCREMENT PRIMARY KEY,
  rfid_tag VARCHAR(20) NOT NULL,
  status TINYINT(1) NULL,
  timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
  INDEX idx_tag_time (rfid_tag, timestamp)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

DELIMITER //
CREATE TRIGGER after_registered_update
AFTER UPDATE ON rfid_registered
FOR EACH ROW
BEGIN
  IF OLD.status <> NEW.status THEN
    INSERT INTO rfid_logs (rfid_tag, status, timestamp)
    VALUES (NEW.rfid_tag, NEW.status, NOW());
  END IF;
END //
DELIMITER ;

DELIMITER //
CREATE TRIGGER after_registered_insert
AFTER INSERT ON rfid_registered
FOR EACH ROW
BEGIN
  INSERT INTO rfid_logs (rfid_tag, status, timestamp)
  VALUES (NEW.rfid_tag, NEW.status, NOW());
END //
DELIMITER ;
